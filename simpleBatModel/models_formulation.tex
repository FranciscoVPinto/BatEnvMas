% ============================================================
% MODEL 1 — BASE SINGLE-HOUSE
% ============================================================
\section{Base Battery Optimisation Model}
\label{sec:base}

\textit{Single-house MILP: load, PV, battery storage, ToU grid tariff,
optional export. No degradation penalty.}

% -------------------
\subsection{Sets and Parameters}

\paragraph{Time}
\[
  t \in \mathcal{T} = \{1, \dots, T\},
  \qquad
  \Delta t \;[\mathrm{h}].
\]

\paragraph{Input profiles}
\begin{itemize}
  \item $L_t$           : load demand $[\mathrm{kW}]$.
  \item $G^{PV}_t$      : available PV generation $[\mathrm{kW}]$.
  \item $c^{grid}_t$    : time-varying grid import price $[\texteuro/\mathrm{kWh}]$.
  \item $c^{sell}_t$    : time-varying export (sell) price $[\texteuro/\mathrm{kWh}]$.
\end{itemize}

\paragraph{Battery parameters}
\begin{itemize}
  \item $P^{ch,\max}$, $P^{dis,\max}$   : maximum charge / discharge power $[\mathrm{kW}]$.
  \item $E^{\min}$, $E^{\max}$           : minimum / maximum stored energy $[\mathrm{kWh}]$.
  \item $\eta^{ch}$, $\eta^{dis} \in (0,1]$ : charge / discharge efficiencies.
  \item $E_0$                             : initial battery energy $[\mathrm{kWh}]$.
  \item $P^{grid,\max}$                  : maximum grid exchange power $[\mathrm{kW}]$.
\end{itemize}

% -------------------
\subsection{Decision Variables}

\begin{itemize}
  \item $P^{ch}_t \ge 0$, $P^{dis}_t \ge 0$ : charge / discharge power $[\mathrm{kW}]$.
  \item $E_t \ge 0$                           : stored energy at end of step $t$ $[\mathrm{kWh}]$.
  \item $x_t \in \{0,1\}$                    : non-simultaneity (1 = charging).
  \item $P^{PV}_t \ge 0$                     : PV power used on-site $[\mathrm{kW}]$.
  \item $P^{curt}_t \ge 0$                   : curtailed PV $[\mathrm{kW}]$.
  \item $P^{imp}_t \ge 0$, $P^{exp}_t \ge 0$ : grid import / export $[\mathrm{kW}]$.
  \item $y_t \in \{0,1\}$                    : non-simultaneity (1 = importing).
\end{itemize}

% -------------------
\subsection{Constraints}

\paragraph{(C0) Initial condition}
\[
  E_0 \;\text{given.}
\]

\paragraph{(C1) Battery non-simultaneity and power limits}
\begin{align}
  0 \le P^{ch}_t   &\le x_t \, P^{ch,\max}       && \forall\,t \in \mathcal{T}, \label{eq:C1_ch} \\
  0 \le P^{dis}_t  &\le (1-x_t)\,P^{dis,\max}    && \forall\,t \in \mathcal{T}. \label{eq:C1_dis}
\end{align}

\paragraph{(C2) Energy dynamics}
\begin{align}
  E_t = E_{t-1}
        + \eta^{ch}\,P^{ch}_t\,\Delta t
        - \frac{P^{dis}_t\,\Delta t}{\eta^{dis}}
  && \forall\,t \in \mathcal{T}. \label{eq:C2}
\end{align}

\paragraph{(C3) Energy bounds}
\begin{align}
  E^{\min} \le E_t \le E^{\max} && \forall\,t \in \mathcal{T}. \label{eq:C3}
\end{align}

\paragraph{(C4) PV balance}
\begin{align}
  P^{PV}_t + P^{curt}_t = G^{PV}_t && \forall\,t \in \mathcal{T}. \label{eq:C4}
\end{align}

\paragraph{(C5) Power balance}
\begin{align}
  L_t = P^{PV}_t + P^{dis}_t - P^{ch}_t + P^{imp}_t - P^{exp}_t
  && \forall\,t \in \mathcal{T}. \label{eq:C5}
\end{align}

\paragraph{(C6) Grid non-simultaneity and limits}
\begin{align}
  0 \le P^{imp}_t &\le y_t \, P^{grid,\max}           && \forall\,t \in \mathcal{T}, \label{eq:C6_imp} \\
  0 \le P^{exp}_t &\le (1-y_t)\,P^{grid,\max}         && \forall\,t \in \mathcal{T}. \label{eq:C6_exp}
\end{align}

\paragraph{(C7) Cyclic SoC (optional, default on)}
\begin{align}
  E_T \ge E_0. \label{eq:C7}
\end{align}
Prevents the optimiser from emptying the battery on the last time step.

% -------------------
\subsection{Objective}
\begin{equation}
  \min \;\sum_{t \in \mathcal{T}}
  \Bigl(c^{grid}_t\,P^{imp}_t - c^{sell}_t\,P^{exp}_t\Bigr)\Delta t.
  \label{eq:OBJ_base}
\end{equation}

% MODEL 2 — MULTI-HOUSE
% ============================================================
\section{Multi-House Model}
\label{sec:multihouse}

\textit{Extension of Model~\ref{sec:base} to $H$ independent households,
each with its own battery, load and PV allocation. All constraints are
replicated per house; no inter-house coupling except through the shared PV
pool (Section~\ref{sec:sharing}).}

% -------------------
\subsection{Additional Sets and Parameters}

\paragraph{Houses}
\[
  h \in \mathcal{H} = \{1, \dots, H\}.
\]

All parameters from Section~\ref{sec:base} become house-indexed:
\[
  L_{h,t},\; G^{PV}_{h,t},\; c^{grid}_{h,t},\; c^{sell}_{h,t},\;
  P^{ch,\max}_h,\; P^{dis,\max}_h,\; E^{\min}_h,\; E^{\max}_h,\;
  \eta^{ch}_h,\; \eta^{dis}_h,\; P^{grid,\max}_h,\; E^{init}_h.
\]

% -------------------
\subsection{Decision Variables}

All variables gain the house index $h$:
\[
  P^{ch}_{h,t},\; P^{dis}_{h,t},\; E_{h,t},\; x_{h,t},\;
  P^{PV}_{h,t},\; P^{curt}_{h,t},\; P^{imp}_{h,t},\; P^{exp}_{h,t},\; y_{h,t}.
\]

% -------------------
\subsection{Constraints}

All constraints~(C1)--(C7) from Section~\ref{sec:base} hold $\forall\,h \in \mathcal{H},\, t \in \mathcal{T}$:

\paragraph{(C1) Battery non-simultaneity}
\begin{align}
  0 \le P^{ch}_{h,t}  &\le x_{h,t}\,P^{ch,\max}_h   && \forall\,h,t, \\
  0 \le P^{dis}_{h,t} &\le (1-x_{h,t})\,P^{dis,\max}_h && \forall\,h,t.
\end{align}

\paragraph{(C2) Energy dynamics}
\begin{align}
  E_{h,t} = E_{h,t-1}
  + \eta^{ch}_h\,P^{ch}_{h,t}\,\Delta t
  - \frac{P^{dis}_{h,t}\,\Delta t}{\eta^{dis}_h}
  && \forall\,h,t.
\end{align}

\paragraph{(C3) Energy bounds}
\begin{align}
  E^{\min}_h \le E_{h,t} \le E^{\max}_h && \forall\,h,t.
\end{align}

\paragraph{(C4) PV balance}
\begin{align}
  P^{PV}_{h,t} + P^{curt}_{h,t} = G^{PV}_{h,t} && \forall\,h,t.
\end{align}

\paragraph{(C5) Power balance}
\begin{align}
  L_{h,t} = P^{PV}_{h,t} + P^{dis}_{h,t} - P^{ch}_{h,t}
           + P^{imp}_{h,t} - P^{exp}_{h,t} && \forall\,h,t.
\end{align}

\paragraph{(C6) Grid limits}
\begin{align}
  0 \le P^{imp}_{h,t} &\le y_{h,t}\,P^{grid,\max}_h   && \forall\,h,t, \\
  0 \le P^{exp}_{h,t} &\le (1-y_{h,t})\,P^{grid,\max}_h && \forall\,h,t.
\end{align}

\paragraph{(C7) Cyclic SoC}
\begin{align}
  E_{h,T} \ge E^{init}_h && \forall\,h.
\end{align}

% -------------------
\subsection{Objective}
\begin{equation}
  \min \;\sum_{h \in \mathcal{H}}\sum_{t \in \mathcal{T}}
  \Bigl(c^{grid}_{h,t}\,P^{imp}_{h,t} - c^{sell}_{h,t}\,P^{exp}_{h,t}\Bigr)\Delta t.
  \label{eq:OBJ_mh}
\end{equation}

% MODEL 3 — MULTI-HOUSE WITH OPTIMAL PV SHARING
% ============================================================
\section{Multi-House Model with Optimal PV Sharing}
\label{sec:sharing}

\textit{Extension of Model~\ref{sec:multihouse} in which the PV allocation
$\alpha_{h,t}$ is a decision variable instead of a fixed heuristic.
The community-level PV total $G^{PV,tot}_t$ is given; the MILP chooses the
allocation that minimises total cost.}

% -------------------
\subsection{Additional Parameter}
\[
  G^{PV,tot}_t = \sum_{h \in \mathcal{H}} G^{PV}_{h,t}
  \;[\mathrm{kW}],
  \quad \forall\,t \in \mathcal{T}.
\]

% -------------------
\subsection{Change: PV as Decision Variable}

In Model~\ref{sec:multihouse}, $G^{PV}_{h,t}$ was an exogenous parameter.
Here it becomes a non-negative variable $P^{PV,alloc}_{h,t}$, and
constraint~(C4) is replaced by:

\paragraph{(C4') PV allocation constraints}
\begin{align}
  \sum_{h \in \mathcal{H}} P^{PV,alloc}_{h,t}
    &= G^{PV,tot}_t                 && \forall\,t \in \mathcal{T}, \label{eq:alpha_sum} \\
  P^{PV,alloc}_{h,t}
    &\le G^{PV,tot}_t               && \forall\,h,t, \label{eq:alpha_bound} \\
  P^{PV,alloc}_{h,t}
    &\ge 0                          && \forall\,h,t. \label{eq:alpha_nonneg}
\end{align}

The implicit sharing fraction is
$\alpha_{h,t} = P^{PV,alloc}_{h,t} / G^{PV,tot}_t$ when $G^{PV,tot}_t > 0$.
All remaining constraints~(C1), (C2), (C3), (C5), (C6), (C7) and the
objective~\eqref{eq:OBJ_mh} remain unchanged, replacing $G^{PV}_{h,t}$ by
$P^{PV,alloc}_{h,t}$.

% -------------------
\subsection{Heuristic Fixed-Alpha Strategies (reference)}

When $P^{PV}_{h,t}$ is a fixed parameter, several deterministic rules for
$\alpha_{h,t}$ are used as benchmarks:

\begin{center}
\begin{tabular}{ll}
\toprule
Strategy & $\alpha_{h,t}$ \\
\midrule
Equal split          & $\alpha_{h,t} = 1/H$ \\
Load-weighted        & $\alpha_{h,t} = L_{h,t}\,/\!\sum_{h'} L_{h',t}$ \\
Consumption-instant  & proportional to consumption at instant $t$ \\
Consumption-mean     & proportional to mean load $\bar{L}_h$ \\
\bottomrule
\end{tabular}
\end{center}

% MODEL 4 — LINEAR BATTERY DEGRADATION
% ============================================================
\section{Multi-House Model with Linear Degradation}
\label{sec:deg_linear}

\textit{Extension of Model~\ref{sec:multihouse} that adds a throughput-based
battery wear cost to the objective. The penalty is proportional to the total
energy cycled through each battery (charge + discharge).}

% -------------------
\subsection{Physical Motivation}

For a battery of capital cost $C_h$ [\texteuro], rated capacity $E^{\max}_h$
[kWh] and cycle life $N_h$, the amortised wear cost per kWh is:
\[
  \lambda^{deg}_h
  = \frac{C_h}{N_h \cdot 2\,E^{\max}_h}
  \;[\texteuro/\mathrm{kWh}].
\]
For a typical 10 kWh LFP pack ($C_h = 4000$\;\texteuro, $N_h = 6000$
cycles): $\lambda^{deg} \approx 0.033$\;\texteuro/kWh.

% -------------------
\subsection{Additional Parameter}
\[
  \lambda^{deg}_h \ge 0
  \;[\texteuro/\mathrm{kWh}] \quad \text{: amortised wear cost per kWh cycled for house } h.
\]

% -------------------
\subsection{Modified Objective}

All constraints remain identical to Model~\ref{sec:multihouse}.
The objective adds a throughput penalty:

\begin{equation}
  \min \;\sum_{h \in \mathcal{H}}\sum_{t \in \mathcal{T}}
  \Bigl[
    \underbrace{\bigl(c^{grid}_{h,t}\,P^{imp}_{h,t}
                     - c^{sell}_{h,t}\,P^{exp}_{h,t}\bigr)\Delta t}_{\text{grid cost}}
    +
    \underbrace{\lambda^{deg}_h
                \bigl(P^{ch}_{h,t} + P^{dis}_{h,t}\bigr)\Delta t}_{\text{degradation cost}}
  \Bigr].
  \label{eq:OBJ_deg}
\end{equation}

The penalty $\lambda^{deg}_h (P^{ch}_{h,t}+P^{dis}_{h,t})\Delta t$ is linear
in the decision variables, so the MILP structure is unchanged.

% MODEL 5 — PWL SoC-DEPENDENT DEGRADATION
% ============================================================
\section{Multi-House Model with PWL Degradation}
\label{sec:deg_pwl}

\textit{Extension of Model~\ref{sec:multihouse} with a piecewise-linear (PWL)
SoC-dependent battery wear cost. Li-Ion and LFP batteries age faster at
extreme SoC; this model captures that asymmetry without leaving the MILP
framework, using binary bin-selection variables and McCormick linearisation.}

% -------------------
\subsection{Physical Motivation}

Three ageing regimes are established in the literature
(Schmalstieg et~al.\ 2014, Hesse et~al.\ 2017):

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{clll}
\toprule
Bin $k$ & SoC range (LFP default) & Mechanism & $\lambda_k$ (default) \\
\midrule
0 & $[0\%, 20\%]$  & Deep-discharge stress, Cu dissolution & 0.08\;\texteuro/kWh \\
1 & $[20\%, 80\%]$ & Preferred operating zone, low SEI growth & 0.03\;\texteuro/kWh \\
2 & $[80\%,100\%]$ & High-voltage overcharge, Li plating      & 0.06\;\texteuro/kWh \\
\bottomrule
\end{tabular}
\end{center}

% -------------------
\subsection{Additional Sets and Parameters}

\paragraph{Bins}
\[
  k \in \mathcal{K} = \{0, \dots, K-1\}.
\]

\paragraph{SoC breakpoints (absolute energy)}

Let $\sigma_0 < \sigma_1 < \cdots < \sigma_K$ be the normalised breakpoints
with $\sigma_0 = 0$ and $\sigma_K = 1$. The absolute energy bounds for bin
$k$ of house $h$ are:
\begin{align}
  e^{lo}_{h,k} &= E^{\min}_h + \sigma_k\,\bigl(E^{\max}_h - E^{\min}_h\bigr), \\
  e^{hi}_{h,k} &= E^{\min}_h + \sigma_{k+1}\,\bigl(E^{\max}_h - E^{\min}_h\bigr).
\end{align}

\paragraph{Big-M constant}
\[
  M_h = E^{\max}_h - E^{\min}_h \;[\mathrm{kWh}].
\]

\paragraph{Degradation cost per bin}
\[
  \lambda_{h,k} \ge 0 \;[\texteuro/\mathrm{kWh}]
  \quad \text{: wear cost for house } h \text{ operating in bin } k.
\]
A global default $\lambda_k$ is used when no per-house override is given.

% -------------------
\subsection{Additional Decision Variables}

\begin{itemize}
  \item $z_{h,t,k} \in \{0,1\}$
        : bin indicator ($z_{h,t,k}=1$ iff battery of house $h$ at time $t$
          is in SoC bin $k$).
  \item $q^{ch}_{h,t,k} \ge 0$
        : McCormick auxiliary for charging in bin $k$ $[\mathrm{kW}]$.
  \item $q^{dis}_{h,t,k} \ge 0$
        : McCormick auxiliary for discharging in bin $k$ $[\mathrm{kW}]$.
\end{itemize}

% -------------------
\subsection{Additional Constraints}

\paragraph{(D1) Unique bin assignment}
\begin{align}
  \sum_{k \in \mathcal{K}} z_{h,t,k} = 1
  && \forall\,h \in \mathcal{H},\, t \in \mathcal{T}.
  \label{eq:D1}
\end{align}

\paragraph{(D2) Big-M bin boundary enforcement}
\begin{align}
  E_{h,t} &\ge e^{lo}_{h,k} - M_h\,(1 - z_{h,t,k})
  && \forall\,h,t,k, \label{eq:D2_lo} \\
  E_{h,t} &\le e^{hi}_{h,k} + M_h\,(1 - z_{h,t,k})
  && \forall\,h,t,k. \label{eq:D2_hi}
\end{align}

\paragraph{(D3) McCormick linearisation of $q^{ch}_{h,t,k} = z_{h,t,k}\cdot P^{ch}_{h,t}$}
\begin{align}
  q^{ch}_{h,t,k} &\le P^{ch,\max}_h\,z_{h,t,k}
  && \forall\,h,t,k, \label{eq:D3_ub_bin} \\
  q^{ch}_{h,t,k} &\le P^{ch}_{h,t}
  && \forall\,h,t,k, \label{eq:D3_ub_pow} \\
  q^{ch}_{h,t,k} &\ge P^{ch}_{h,t} - P^{ch,\max}_h\,(1-z_{h,t,k})
  && \forall\,h,t,k. \label{eq:D3_lb}
\end{align}

\paragraph{(D4) McCormick linearisation of $q^{dis}_{h,t,k} = z_{h,t,k}\cdot P^{dis}_{h,t}$}
\begin{align}
  q^{dis}_{h,t,k} &\le P^{dis,\max}_h\,z_{h,t,k}
  && \forall\,h,t,k, \label{eq:D4_ub_bin} \\
  q^{dis}_{h,t,k} &\le P^{dis}_{h,t}
  && \forall\,h,t,k, \label{eq:D4_ub_pow} \\
  q^{dis}_{h,t,k} &\ge P^{dis}_{h,t} - P^{dis,\max}_h\,(1-z_{h,t,k})
  && \forall\,h,t,k. \label{eq:D4_lb}
\end{align}

The McCormick envelope gives the tightest linear relaxation of the bilinear
product $z_{h,t,k}\cdot P_{h,t}$ when $z \in \{0,1\}$, and is exact at the
integer vertices (McCormick 1976).

% -------------------
\subsection{Modified Objective}

All constraints from Model~\ref{sec:multihouse} plus (D1)--(D4) apply.
The objective replaces the flat degradation term with the SoC-dependent one:

\begin{equation}
  \min \;\sum_{h \in \mathcal{H}}\sum_{t \in \mathcal{T}}
  \left[
    \bigl(c^{grid}_{h,t}\,P^{imp}_{h,t}
         - c^{sell}_{h,t}\,P^{exp}_{h,t}\bigr)\Delta t
    +
    \sum_{k \in \mathcal{K}}
    \lambda_{h,k}\,\bigl(q^{ch}_{h,t,k} + q^{dis}_{h,t,k}\bigr)\,\Delta t
  \right].
  \label{eq:OBJ_pwl}
\end{equation}

Because $z_{h,t,k} \in \{0,1\}$ and $q = z \cdot P$ via
constraints~\eqref{eq:D3_ub_bin}--\eqref{eq:D3_lb} and
\eqref{eq:D4_ub_bin}--\eqref{eq:D4_lb}, the term
$\lambda_{h,k}\,q^{ch}_{h,t,k}$ evaluates to
$\lambda_{h,k}\,P^{ch}_{h,t}$ when bin $k$ is active and to $0$ otherwise.
The PWL cost is therefore:
\[
  \text{deg cost}_{h,t}
  = \sum_{k} \lambda_{h,k}\,z_{h,t,k}\,P^{ch/dis}_{h,t}\,\Delta t
  = \lambda_{h,k^*}\,(P^{ch}_{h,t}+P^{dis}_{h,t})\,\Delta t,
\]
where $k^*$ is the unique active bin.

% -------------------
\subsection{Complexity Comparison}

\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{lcccc}
\toprule
Model & Binary vars & Continuous vars & Constraints & MILP structure \\
\midrule
Base (M1)            & $2HT$         & $7HT + HT_E$   & $\mathcal{O}(HT)$    & LP relaxation tight \\
Multi-house (M2)     & $2HT$         & $7HT + HT_E$   & $\mathcal{O}(HT)$    & same \\
Optimal sharing (M3) & $2HT$         & $8HT + HT_E$   & $\mathcal{O}(HT)$    & adds $T$ eq. \\
Linear deg.\ (M4)    & $2HT$         & $7HT + HT_E$   & $\mathcal{O}(HT)$    & objective only \\
PWL deg.\ (M5)       & $(2+K)HT$     & $(7+2K)HT + HT_E$ & $\mathcal{O}(KHT)$ & $6K+1$ extra /$(h,t)$ \\
\bottomrule
\end{tabular}
\end{center}

For $K=3$, $H=5$, $T=96$ (one day, 15\,min resolution):
M5 adds $+1\,440$ binary and $+2\,880$ continuous variables over M4.
